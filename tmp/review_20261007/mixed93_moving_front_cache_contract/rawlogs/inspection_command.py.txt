def sha(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()

def archive_headers(path, wanted):
    out = {}
    with zipfile.ZipFile(path, "r") as zf:
        files = {n[:-4]: n for n in zf.namelist() if n.endswith(".npy")}
        for key in wanted:
            member = files.get(key)
            if member is None:
                continue
            with zf.open(member, "r") as f:
                if f.read(6) != b"\x93NUMPY":
                    raise ValueError("bad NPY magic: " + key)
                major, minor = f.read(2)
                hlen = int.from_bytes(f.read(2 if major == 1 else 4), "little")
                header = ast.literal_eval(f.read(hlen).decode("latin1").strip())
            out[key] = {"shape": list(header["shape"]), "dtype": header["descr"], "fortran_order": bool(header["fortran_order"])}
        return len(files), files, out

def text_list(a):
    vals = []
    for v in np.asarray(a).reshape(-1):
        vals.append(v.decode("utf-8") if isinstance(v, bytes) else str(v))
    return vals

def hbytes(a):
    return hashlib.sha256(np.ascontiguousarray(a).tobytes()).hexdigest()

def cloud(prefix, z, unique):
    xyz = np.asarray(z[prefix + "_xyz_m"])
    ids = np.asarray(z[prefix + "_unique_row_ids"])
    wkey = prefix + "_weights"
    weights = np.asarray(z[wkey]) if wkey in z.files else None
    valid = bool(ids.ndim == 1 and len(ids) == len(xyz) and ids.size > 0 and ids.min() >= 0 and ids.max() < len(unique))
    matches = bool(valid and np.array_equal(xyz, unique[ids]))
    return {"xyz_key": prefix + "_xyz_m", "row_id_key": prefix + "_unique_row_ids", "weight_key": wkey if weights is not None else None,
            "row_count": int(len(xyz)), "xyz_shape": list(xyz.shape), "row_id_dtype": str(ids.dtype),
            "row_id_min": int(ids.min()) if ids.size else None, "row_id_max": int(ids.max()) if ids.size else None,
            "unique_row_id_count": int(np.unique(ids).size), "xyz_equals_unique_xyz_indexed_by_row_ids": matches,
            "xyz_sha256_float64_c_order": hbytes(xyz), "row_ids_sha256": hbytes(ids),
            "weights_shape": list(weights.shape) if weights is not None else None,
            "weights_sha256": hbytes(weights) if weights is not None else None}
